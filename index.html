break; } f[count++] = s.substring(start, comma); start = comma + 1; } portENTER_CRITICAL(&mux); if(s.startsWith("$GNGGA")) { int quality = f[6].toInt(); if(quality > 0) { gpsFix = true; latitude = nmeaToDecimal(f[2], f[3]); } int star = s.indexOf('*'); if(star > 0) s = s.substring(0, star); String f[15]; int start = 0, count = 0; while(count < 15) { int comma = s.indexOf(',', start); if(comma < 0) { f[count++] = s.substring(start); break; } f[count++] = s.substring(start, comma); start = comma + 1; } portENTER_CRITICAL(&mux); if(s.startsWith("$WIMWV")) { windDir = f[1].toFloat(); windSpeed = f[3].toFloat(); } portEXIT_CRITICAL(&mux); } void readBluetooth() { while(SerialBT.available()) { char c = SerialBT.read(); if(c == '\r') continue; if(c == '\n') { if(btBuffer.length() > 5) parseMiniPlexNMEA(btBuffer); btBuffer = ""; } else { btBuffer += c; gpsBuffer = ""; } else { gpsBuffer += c; if(gpsBuffer.length() > 120) gpsBuffer = ""; } } } //================================================== // LECTURA BATERIAS //================================================== void readBatteries() { batMotor = analogRead(PIN_BAT_MOTOR) * 3.3 / 4095.0 * 12.8; } //================================================== // PUBLICACION MQTT //================================================== void publishData() { StaticJsonDocument<1024> doc; portENTER_CRITICAL(&mux); portEXIT_CRITICAL(&mux); doc["bm"] = round(batMotor * 10.0) / 10.0; // bm // tension bateria del motor doc["bs"] = round(batService * 10.0) / 10.0; // bs // tension bateria de servicio doc["bb"] = round(batBow * 10.0) / 10.0; Serial.print(topicData); Serial.print("]: "); Serial.println(buffer); } //================================================== // WIFI NO BLOQUEANTE //================================================== void connectWiFi() { if(WiFi.status() == WL_CONNECTED) { if(wifiStatus != "OK") { wifiStatus = "OK";
